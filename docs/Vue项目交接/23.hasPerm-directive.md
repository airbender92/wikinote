# 这是什么？
这是 Vue **自定义指令 v-hasPermi**
用来做**按钮级权限控制**，比如：
```vue
<el-button v-hasPermi="'system:user:add'">新增用户</el-button>
```
没有权限时，**按钮会直接从 DOM 里删掉**。

# 完整注释版
```ts
import type { App, Directive, DirectiveBinding } from 'vue'
// 权限判断工具方法
import { hasPermission } from '@/utils/auth'

// 核心逻辑：判断是否有权限
function hasPermi(el: Element, binding: DirectiveBinding) {
  // 去掉指令里可能带的单引号
  const value = binding.value.replace(/'/g, '')

  // 调用权限判断方法
  const flag = hasPermission(value)

  // 没权限 → 直接把当前元素从 DOM 移除
  if (!flag) {
    el.parentNode?.removeChild(el)
  }
}

// 挂载时执行
const mounted = (el: Element, binding: DirectiveBinding<any>) => {
  hasPermi(el, binding)
}

// 更新时再次执行（权限变化时）
const updated = (el: Element, binding: DirectiveBinding<any>) => {
  hasPermi(el, binding)
}

// 定义指令
const permiDirective: Directive = {
  mounted,
  updated
}

// 注册到 Vue 实例
export const setupPermissionDirective = (app: App<Element>) => {
  app.directive('hasPermi', permiDirective)
}

export default permiDirective
```

# 核心逻辑超简单
1. 页面渲染时执行 `hasPermi`
2. 调用 `hasPermission(value)` 判断当前用户**是否拥有该权限码**
3. 没权限：
   ```ts
   el.parentNode?.removeChild(el)
   ```
   直接把按钮**从 DOM 里删除**，不是隐藏，是真·删除。
