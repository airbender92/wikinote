export const generateRoutesByFrontEnd = (
  routes: AppRouteRecordRaw[],
  keys: string[],
  basePath = '/'
): AppRouteRecordRaw[] => {
  const res: AppRouteRecordRaw[] = []

  for (const route of routes) {
    const meta = route.meta ?? {}
    // hidden 且不能跳转的直接跳过
    if (meta.hidden && !meta.canTo) {
      continue
    }

    let data: Nullable<AppRouteRecordRaw> = null

    const userStore = useUserStoreWithOut()
    const userPermissions = userStore.getPermissions
    const isAdmin = userStore.getIsAdmin

    const routePermissions = meta.permission

    // 满足任一条件直接显示路由：
    // 1. 全局没开权限控制
    // 2. 是管理员
    // 3. 路由没配权限
    // 4. alwaysShow 强制显示
    if (
      import.meta.env.VITE_USE_PERMISSION !== 'true' ||
      isAdmin ||
      !routePermissions ||
      routePermissions.length === 0 ||
      meta.alwaysShow
    ) {
      data = Object.assign({}, route)
    } else {
      // 判断用户权限是否包含路由需要的权限
      const hasPermission = routePermissions.some((perm) => userPermissions.includes(perm))
      if (hasPermission) {
        data = Object.assign({}, route)
      }
    }

    // 递归处理子路由
    if (route.children && data) {
      data.children = generateRoutesByFrontEnd(
        route.children,
        keys,
        pathResolve(basePath, data.path)
      )
    }
    if (data) {
      res.push(data as AppRouteRecordRaw)
    }
  }
  return res
}

export const generateRoutesByServer = (routes: AppCustomRouteRecordRaw[]): AppRouteRecordRaw[] => {
  const res: AppRouteRecordRaw[] = []

  for (const route of routes) {
    const data: AppRouteRecordRaw = {
      path: route.path,
      name: route.name,
      redirect: route.redirect,
      meta: route.meta
    }
    if (route.component) {
      // 去前面 glob 导入的模块里找对应组件
      const comModule = modules[`../${route.component}.vue`] || modules[`../${route.component}.tsx`]
      const component = route.component as string

      // 没找到文件打印警告
      if (!comModule && !component.includes('#')) {
        console.error(`未找到${route.component}.vue文件或${route.component}.tsx文件，请创建`)
      } else {
        // 组件映射规则：
        // # → Layout
        // ## → ParentLayout
        // 其他 → 对应页面组件
        data.component =
          component === '#' ? Layout : component.includes('##') ? getParentLayout() : comModule
      }
    }
    // 递归子路由
    if (route.children) {
      data.children = generateRoutesByServer(route.children)
    }
    res.push(data as AppRouteRecordRaw)
  }
  return res
}

// 拼接父子路由路径，处理 // 变 /
export const pathResolve = (parentPath: string, path: string) => {
  if (isUrl(path)) return path
  const childPath = path.startsWith('/') || !path ? path : `/${path}`
  return `${parentPath}${childPath}`.replace(/\/\//g, '/').trim()
}

const isMultipleRoute = (route: AppRouteRecordRaw) => {
  if (!route || !Reflect.has(route, 'children') || !route.children?.length) {
    return false
  }

  const children = route.children

  let flag = false
  for (let index = 0; index < children.length; index++) {
    const child = children[index]
    // 只要有一个子路由还有 children，就是多级
    if (child.children?.length) {
      flag = true
      break
    }

  return flag

// 核心：把多级路由提升为二级路由
const promoteRouteLevel = (route: AppRouteRecordRaw) => {
  // 临时创建 router 实例，用于解析完整路由
  let router: Router | null = createRouter({
    routes: [route as RouteRecordRaw],
    history: createWebHashHistory()
  })

  const routes = router.getRoutes()
  // 把所有深层子路由提升到一级 children
  addToChildren(routes, route.children || [], route)
  router = null

  // 去掉子路由里的 children，变成纯二级路由
  route.children = route.children?.map((item) => omit(item, 'children'))
}

// 递归把所有深层子路由添加到一级 children
const addToChildren = (
  routes: RouteRecordNormalized[],
  children: AppRouteRecordRaw[],
  routeModule: AppRouteRecordRaw
) => {
  for (let index = 0; index < children.length; index++) {
    const child = children[index]
    const route = routes.find((item) => item.name === child.name)
    if (!route) {
      continue
    }
    routeModule.children = routeModule.children || []
    if (!routeModule.children.find((item) => item.name === route.name)) {
      routeModule.children?.push(route as unknown as AppRouteRecordRaw)
    }
    // 孙子路由也拉上来
    if (child.children?.length) {
      addToChildren(routes, child.children, routeModule)
    }
  }
}

# 先说两个关键澄清，纠正一句流传很广的误区
> ❌ 错误表述：Vue keep-alive 不支持三级路由
> ✅ 真相：**keep-alive 本身支持无限层级组件缓存**。
> 真正出问题的场景 = **后台管理系统「Layout + 标签页(Tabs) + 顶层唯一 `<router-view>`」这套架构**。

你源码里这套 `flatMultiLevelRoutes` 拍平逻辑，是 **vue-element-admin/vue-vben-admin 一脉传承的折中方案**，专门适配「多标签页系统」。

## 一、先回答：三级路由不缓存，现实表现是什么？
举标准后台场景：
```
/ system (Layout)
    / user (二级，目录占位组件 ParentLayout)
        / list (三级：用户列表页)
```
业务行为：
1. 用户打开【用户列表】，设置筛选条件、翻到第5页
2. 点开其他菜单，再切回【用户列表】

### 如果三级路由缓存失效会发生：
- `onMounted` **重复执行**，接口重新请求
- 分页、筛选条件、表单输入、滚动位置全部丢失
- 用户需要重新勾选筛选、翻页，体验极差

底层原理（对应你项目架构）
顶层 Layout 只有**一处**：
```vue
<!-- Layout.vue -->
<keep-alive :include="cachedViews">
  <router-view />
</keep-alive>
```
路由结构渲染链路：
Layout(缓存容器) → ParentLayout(中间占位路由) → UserList(页面)

1. `keep-alive` 缓存直接子组件：也就是 `ParentLayout`
2. 当你离开这个路由，**中间层 ParentLayout 一旦被销毁**
3. 内部嵌套 `<router-view>` 连同子页面缓存**全部销毁**
> 覆巢之下无完卵。
哪怕 UserList 写了 `meta.keepAlive=true`，缓存也保不住。

> 注意：
> 如果不用标签页、没有多层中间占位组件，只是普通网站嵌套路由，**三级路由缓存完全没问题**。问题高度绑定【后台管理系统架构】。

## 二、第二个问题：现实中有没有深层嵌套路由的需求？
**大量存在。**
1. 大型ERP、生产管理、OA，菜单层级可以做到4~5级
```
生产管理
    工单管理
        车间工单
            工单明细
                工序记录
```
产品侧希望菜单树严格按照业务层级展示，方便新人理解业务归属。

### 矛盾来了：
✅ **菜单展示：需要多层嵌套，还原业务层级**
❌ **路由运行：多层嵌套会引发 keep-alive 缓存不稳定**

现在出现两条技术路线，对应市面上两套方案：
## 方案A（你当前源码采用：vben-admin思路）
1. **原始路由表：保留多层嵌套** → 用来渲染左侧菜单，树形结构正常展示
2. **给vue-router注册的最终路由：执行 flatMultiLevelRoutes 拍平成二级**
```
生产管理（一级Layout）
├─工单管理
├─车间工单
├─工单明细
└─工序记录
```
- 路由路径不变 `/production/workshop/order/detail`
- 路由嵌套结构变成扁平二级
- 菜单渲染使用**未拍平的原始路由数组**
> 核心技巧：两份路由表分开使用！
> - 菜单渲染：原始嵌套路由
> - 注册给router、页面缓存：拍平后的二级路由

优点：
- keep-alive 缓存稳定，极少出现玄学bug
- 不用多层嵌套 keep-alive、不用维护复杂缓存链路
缺点：
- 需要维护两套路由数据，理解门槛高（也就是你现在正在看的一堆工具函数）

## 方案B（不拍平路由，保留原生嵌套）
不修改路由结构，解决缓存问题，两条实现分支：
1. **ParentView 嵌套缓存方案**
每一层中间目录组件内部都包裹一层 `<keep-alive>`，逐级传递缓存。
缺点：层级越深，缓存链路越复杂；关闭标签页清除缓存时容易遗留内存。

2. **路由守卫改写 matched 数组**
在 `beforeEach` 手动篡改 `to.matched`，欺骗顶层 keep-alive，模拟二级路由。
缺点：操作vue-router内部只读对象，vue-router版本升级容易break，属于hack。

## 四、延伸：什么时候不建议用路由拍平？
1. 项目路由**全部是动态后端路由**，大量详情页动态路由（/:id）
2. 团队维护人员不清楚「两份路由表」的区别，极易混淆造成bug
3. 需要大量路由守卫依赖路由父子嵌套关系做业务判断（权限继承、面包屑）

面包屑注意坑：
路由拍平之后，router内的 `matched` 是扁平结构。
如果你面包屑依赖路由嵌套层级自动生成，**不能使用拍平后的路由**，依然要基于原始嵌套路由去构建面包屑！这也是绝大多数人踩坑的地方。
