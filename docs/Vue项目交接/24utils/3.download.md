# 完整注释版
```typescript
import { ElMessage } from 'element-plus'
import { useI18n } from '@/hooks/web/useI18n'

const { t } = useI18n()

/**
 * 从响应头 Content-Disposition 中提取文件名
 */
const extractFilenameFromHeaders = (headers: any): string | null => {
  if (!headers) return null

  const contentDisposition = headers['content-disposition'] || headers['Content-Disposition']
  if (!contentDisposition) return null

  // 匹配 filename=xxx
  const match = contentDisposition.match(/filename[^;=\n]*=((['"]).*?\2|[^;\n]*)/)
  if (!match) return null

  let filename = match[1]
  // 去掉引号
  if (
    (filename.startsWith('"') && filename.endsWith('"')) ||
    (filename.startsWith("'") && filename.endsWith("'"))
  ) {
    filename = filename.slice(1, -1)
  }

  // 解码 URL 编码文件名
  try {
    return decodeURIComponent(filename)
  } catch {
    return filename
  }
}

/**
 * 解析 Blob 里的 JSON 错误信息
 * 后端下载接口报错时，经常返回 JSON 格式的 Blob
 */
const parseBlobErrorMessage = async (blob: Blob): Promise<string | null> => {
  try {
    const text = await blob.text()
    if (!text) return null

    const json = JSON.parse(text)
    return json?.msg || json?.message || null
  } catch {
    return null
  }
}

type SaveWithPickerResult = 'saved' | 'cancelled' | 'unsupported'

/**
 * 优先使用浏览器原生文件选择器保存（现代 Chrome/Edge 支持）
 */
const saveBlobWithPicker = async (blob: Blob, fileName: string): Promise<SaveWithPickerResult> => {
  const showSaveFilePicker = (window as any).showSaveFilePicker

  // 不支持新 API 则回退
  if (typeof showSaveFilePicker !== 'function') {
    return 'unsupported'
  }

  try {
    const ext = fileName.includes('.') ? `.${fileName.split('.').pop()}` : ''
    const fileHandle = await showSaveFilePicker({
      suggestedName: fileName,
      types: ext
        ? [
            {
              description: 'File',
              accept: { [blob.type || 'application/octet-stream']: [ext] }
            }
          ]
        : undefined
    })

    const writable = await fileHandle.createWritable()
    await writable.write(blob)
    await writable.close()

    return 'saved'
  } catch (err: any) {
    // 用户手动取消，不提示错误
    if (err?.name === 'AbortError') return 'cancelled'
    throw err
  }
}

/**
 * 对外核心方法：下载 Blob 文件
 */
export const downloadFile = async (
  response: any,
  fileName: string,
  successMessage: string = t('common.downloadSuccess'),
  errorMessage: string = t('common.downloadFailed')
): Promise<boolean> => {
  try {
    if (!response) {
      ElMessage.error(errorMessage)
      return false
    }

    // 兼容直接传 Blob / Axios 响应 { data: Blob }
    const actualBlob =
      response instanceof Blob ? response : response.data instanceof Blob ? response.data : null

    if (!actualBlob) {
      console.error('无效的 Blob 数据', response)
      ElMessage.error(errorMessage)
      return false
    }

    const headers = response?.headers || {}
    const contentType = (headers['content-type'] || headers['Content-Type'] || actualBlob.type || '').toLowerCase()

    // 如果返回的是 JSON 错误（而非文件流），解析并提示
    if (contentType.includes('application/json') || contentType.includes('text/json')) {
      const msg = await parseBlobErrorMessage(actualBlob)
      ElMessage.error(msg || errorMessage)
      return false
    }

    // 从 header 读取真实文件名
    const finalFileName = extractFilenameFromHeaders(headers) || fileName

    // 优先使用原生文件选择器保存
    const saveResult = await saveBlobWithPicker(actualBlob, finalFileName)
    if (saveResult === 'saved') {
      successMessage && ElMessage.success(successMessage)
      return true
    }
    if (saveResult === 'cancelled') {
      return false
    }

    // 不支持新 API → 降级使用传统 a 标签下载
    const url = URL.createObjectURL(actualBlob)
    const a = document.createElement('a')
    a.href = url
    a.download = finalFileName
    a.click()
    URL.revokeObjectURL(url)

    successMessage && ElMessage.success(successMessage)
    return true

  } catch (err) {
    console.error('下载失败', err)
    ElMessage.error(errorMessage)
    return false
  }
}

/**
 * 简化模板下载调用
 */
export const handleTemplateDownload = async (
  apiCall: () => Promise<any>,
  fileName: string,
  successMessage?: string,
  errorMessage?: string
): Promise<boolean> => {
  try {
    const blobRes = await apiCall()
    return downloadFile(blobRes, fileName, successMessage, errorMessage)
  } catch (err) {
    console.error('模板下载失败', err)
    ElMessage.error(errorMessage || t('common.downloadFailed'))
    return false
  }
}
```

# 核心亮点（为什么这段写得很专业）
## 1. 兼容两种下载方式
- **现代浏览器**：使用 `showSaveFilePicker` 弹出系统保存对话框，体验更好
- **旧浏览器 / 不支持**：自动降级用 `<a>` 标签下载

## 2. 自动从响应头读取文件名
后端在 `Content-Disposition: filename="xxx.xlsx"`
工具会**自动解析并使用**，优先级高于前端传入的 fileName

## 3. 处理后端下载报错（非常关键）
很多项目的坑：
导出接口报错时，后端依然返回 `blob`，但内容是 JSON 错误信息
这段代码会：
- 判断 `content-type: application/json`
- 解析 `msg/message`
- 弹出错误提示，**不会下载成一个损坏文件**

## 4. 国际化、异常捕获、用户体验完整
- 支持 i18n 提示
- 用户取消保存不弹错误
- 成功/失败统一提示
- 完整 try/catch

# 实际业务怎么用
## ① 普通导出/下载
```ts
import { downloadFile } from '@/utils/download'

export const exportUserList = (params) => {
  return request.post({
    url: '/system/user/export',
    params,
    responseType: 'blob'
  })
}

// 页面调用
const handleExport = async () => {
  const res = await exportUserList(params)
  downloadFile(res, '用户列表.xlsx')
}
```

## ② 模板下载（更简洁）
```ts
import { handleTemplateDownload } from '@/utils/download'

const downloadTemplate = () => {
  handleTemplateDownload(() => 
    request.get({ url: '/system/user/template', responseType: 'blob' })
  , '用户导入模板.xlsx')
}
```
