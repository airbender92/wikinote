```typescript
import axios, { AxiosError } from 'axios'
import { defaultRequestInterceptors, defaultResponseInterceptors } from './config'

import {
  AxiosInstance,
  InternalAxiosRequestConfig,
  RequestConfig,
  AxiosResponse
} from './types'
import { ElMessage } from 'element-plus'
import { REQUEST_TIMEOUT } from '@/constants'
import { store } from '@/store'
import { useUserStoreWithOut } from '@/store/modules/user'
import router from '@/router'
import { isUnauthorizedResponse, shouldSkipAuthHandling } from './auth'

// 接口基础地址
export const PATH_URL = import.meta.env.VITE_API_BASE_PATH

// 用于取消请求的控制器
const abortControllerMap: Map<string, AbortController> = new Map()

// token 续签响应头字段
const TOKEN_REFRESH_HEADER = 'x-refresh-token'
// 未授权重定向缓存 key
const UN_AUTH_REDIRECT_CACHE_KEY = 'UN_AUTH_REDIRECT'
// 防止重复处理 401
let isHandlingUnauthorized = false

// 判断是否需要续签 token（登录/登出接口不续签）
export const shouldRefreshToken = (url?: string) => {
  if (!url) return false
  return !shouldSkipAuthHandling(url)
}

// 删除对应请求的取消控制器
export const deleteAbortController = (url?: string) => {
  if (!url) return
  abortControllerMap.delete(url)
}

// 从响应头获取新 token（兼容大小写、AxiosHeaders 格式）
export const getRefreshTokenFromHeaders = (headers?: Record<string, any>) => {
  if (!headers) return ''

  if (typeof headers.get === 'function') {
    return headers.get(TOKEN_REFRESH_HEADER) || headers.get('X-Refresh-Token') || ''
  }

  return headers[TOKEN_REFRESH_HEADER] || headers['X-Refresh-Token'] || ''
}

// 同步刷新本地 token
export const syncRefreshToken = (response?: Pick<AxiosResponse, 'config' | 'headers'>) => {
  const requestUrl = response?.config?.url
  if (!shouldRefreshToken(requestUrl)) return

  const refreshToken = getRefreshTokenFromHeaders(response?.headers as Record<string, any>)
  if (!refreshToken) return

  // 后端返回新 token 就自动更新本地
  const userStore = useUserStoreWithOut()
  userStore.setToken(refreshToken)
}

// 取消所有请求
export const cancelAllRequest = () => {
  for (const [_, controller] of abortControllerMap) {
    controller.abort()
  }
  abortControllerMap.clear()
}

// 重置所有 pinia 状态
export const resetAllStores = () => {
  ;(store as any)._s.forEach((piniaStore: { $reset?: () => void }) => {
    piniaStore.$reset?.()
  })
}

// 构建登录页跳转地址（带 redirect 回跳参数）
export const getLoginRedirectHref = () => {
  const currentRoute = router.currentRoute.value
  const userStore = useUserStoreWithOut()

  const configuredRedirectUrl = String(userStore.getRedirectUrl || '').trim()
  const sessionCachedRedirectUrl = String(
    window.sessionStorage.getItem(UN_AUTH_REDIRECT_CACHE_KEY) || ''
  ).trim()

  const redirectUrl =
    configuredRedirectUrl && configuredRedirectUrl !== '/login'
      ? configuredRedirectUrl
      : sessionCachedRedirectUrl || configuredRedirectUrl

  if (!redirectUrl) {
    return router.resolve({ path: '/login' }).href
  }

  const redirectPath = currentRoute.fullPath || currentRoute.path || '/'

  // 外部链接
  if (/^https?:\/\//i.test(redirectUrl)) {
    try {
      const targetUrl = new URL(redirectUrl)
      targetUrl.searchParams.set('redirect', redirectPath)
      return targetUrl.toString()
    } catch {
      return redirectUrl
    }
  }

  // 内部路由
  const resolvedRedirect = router.resolve({
    path: redirectUrl,
    query: { redirect: redirectPath }
  })

  return resolvedRedirect.href
}

// 401 统一处理：清缓存 + 跳登录页
export const handleUnauthorized = (url?: string) => {
  if (shouldSkipAuthHandling(url)) return false
  if (isHandlingUnauthorized) return true

  isHandlingUnauthorized = true
  const redirectUrl = getLoginRedirectHref()

  // 取消所有请求 + 清空状态 + 清空存储
  cancelAllRequest()
  resetAllStores()
  window.localStorage.clear()
  window.sessionStorage.clear()

  // iframe 场景跳到顶层
  if (window.top !== window.self) {
    window?.top?.location.reload()
  } else {
    window.location.href = redirectUrl
  }

  return true
}

// 判断是否是主动取消请求
export const isCanceledError = (error: AxiosError) => {
  return error.code === 'ERR_CANCELED' || error.name === 'CanceledError'
}

// 解析请求错误信息
export const resolveAxiosErrorMessage = (error: AxiosError) => {
  const candidateList = [error.message, error.response?.statusText]

  for (const candidate of candidateList) {
    const text = String(candidate ?? '').trim()
    if (text) return text
  }

  return 'Request failed'
}

// 创建 axios 实例
const axiosInstance: AxiosInstance = axios.create({
  timeout: REQUEST_TIMEOUT,
  baseURL: PATH_URL
})

// 请求拦截器：自动添加取消控制器
axiosInstance.interceptors.request.use((res: InternalAxiosRequestConfig) => {
  const controller = new AbortController()
  const url = res.url || ''
  res.signal = controller.signal

  // mock 模式去掉 /mock 前缀
  abortControllerMap.set(
    import.meta.env.VITE_USE_MOCK === 'true' ? url.replace('/mock', '') : url,
    controller
  )
  return res
})

// 响应拦截器：处理取消、token 续签、401、错误提示
axiosInstance.interceptors.response.use(
  (res: AxiosResponse) => {
    const url = res.config.url || ''
    deleteAbortController(url)
    // 自动续签 token
    syncRefreshToken(res)

    // 401 处理
    if (isUnauthorizedResponse(res)) {
      handleUnauthorized(url)
    }

    return res
  },
  (error: AxiosError) => {
    deleteAbortController(error.config?.url)
    syncRefreshToken(error.response as AxiosResponse | undefined)

    // 请求被取消，直接 reject
    if (isCanceledError(error)) {
      return Promise.reject(error)
    }

    // 401 处理
    if (isUnauthorizedResponse(error.response as AxiosResponse | undefined)) {
      if (handleUnauthorized(error.config?.url)) {
        return Promise.reject(error)
      }
    }

    console.error('err：', error)
    ElMessage.error(resolveAxiosErrorMessage(error))
    return Promise.reject(error)
  }
)

// 挂载通用请求/响应处理（表单、formData、get 防缓存、业务 code 判断）
axiosInstance.interceptors.request.use(defaultRequestInterceptors)
axiosInstance.interceptors.response.use(defaultResponseInterceptors)

// 最终导出的 service
const service = {
  request: (config: RequestConfig) => {
    return new Promise((resolve, reject) => {
      // 支持单个请求自定义拦截器
      if (config.interceptors?.requestInterceptors) {
        config = config.interceptors.requestInterceptors(config as any)
      }

      axiosInstance
        .request(config)
        .then((res) => {
          resolve(res)
        })
        .catch((err) => {
          reject(err)
        })
    })
  },
  cancelRequest: (url: string | string[]) => {
    const urlList = Array.isArray(url) ? url : [url]
    for (const _url of urlList) {
      abortControllerMap.get(_url)?.abort()
      abortControllerMap.delete(_url)
    }
  },
  cancelAllRequest() {
    cancelAllRequest()
  }
}

export default service
```

不会覆盖，**多个 `use` 会按顺序依次执行，是叠加关系，不是覆盖**。

我给你用最简单的话讲清楚：

# 1. 结论
```ts
axiosInstance.interceptors.request.use(拦截器A)
axiosInstance.interceptors.request.use(拦截器B)
```

结果：
**请求会先执行 A，再执行 B，两个都会生效**
不是后面覆盖前面。

---

# 2. 你这段代码实际执行顺序
```ts
// 拦截器1：处理取消请求、AbortController
axiosInstance.interceptors.request.use((config) => {
  // 加 abort 控制器
  return config
})

// 拦截器2：处理表单、form-data、get 加时间戳
axiosInstance.interceptors.request.use(defaultRequestInterceptors)
```

真实执行顺序：
1. 先进入 **拦截器1**（添加 abort 信号）
2. 再进入 **拦截器2**（处理表单/参数）
3. 最后才发送请求

两个都跑，**互不覆盖**。

---

# 3. 为什么不会覆盖？
Axios 的拦截器内部是一个**数组**
每次 `use()` 都是往数组里 `push` 一个函数。

```
interceptors.request = [拦截器1, 拦截器2]
```

请求时会**循环执行一遍**。

---

# 4. 唯一要注意的是：顺序
你现在的顺序是：

1. abort 控制器
2. 业务处理（表单、GET 时间戳）

这是**完全合理、正确**的顺序。

如果顺序反过来也能用，只是逻辑没那么干净，但不会报错。

多个 `interceptors.request.use`
**不会覆盖，会依次执行，顺序执行，全部生效**。
