# 完整带注释版
```typescript
// 引入底层 axios 实例（service.ts）
import service from './service'
import { CONTENT_TYPE } from '@/constants'
// 获取用户 token
import { useUserStoreWithOut } from '@/store/modules/user'

/**
 * 底层 request 核心方法
 */
const request = (option: AxiosConfig, baseURL?: string) => {
  const { url, method, params, data, headers, responseType } = option

  const userStore = useUserStoreWithOut()
  const token = userStore.getToken

  // 自动拼接 token 请求头
  const tokenHeaders = token
    ? {
        [userStore.getTokenKey ?? 'Authorization']: token
      }
    : {}

  return service.request({
    // 接口基础地址，优先传参，否则用环境变量 VITE_API_BASE_PATH
    baseURL: baseURL ?? import.meta.env.VITE_API_BASE_PATH,
    url,
    method,
    params,
    data,
    responseType,
    // 请求头：默认 Content-Type + token + 自定义 headers
    headers: {
      'Content-Type': CONTENT_TYPE,
      ...tokenHeaders,
      ...headers
    }
  })
}

// 对外暴露的常用方法
export default {
  get: <T = any>(option: AxiosConfig, baseURL?: string) => {
    return request({ method: 'get', ...option }, baseURL) as Promise<IResponse<T>>
  },
  post: <T = any>(option: AxiosConfig, baseURL?: string) => {
    return request({ method: 'post', ...option }, baseURL) as Promise<IResponse<T>>
  },
  delete: <T = any>(option: AxiosConfig, baseURL?: string) => {
    return request({ method: 'delete', ...option }, baseURL) as Promise<IResponse<T>>
  },
  put: <T = any>(option: AxiosConfig, baseURL?: string) => {
    return request({ method: 'put', ...option }, baseURL) as Promise<IResponse<T>>
  },

  // 取消单个/全部请求
  cancelRequest: (url: string | string[]) => {
    return service.cancelRequest(url)
  },
  cancelAllRequest: () => {
    return service.cancelAllRequest()
  }
}
```

# 核心功能一句话总结
- 所有接口**自动携带 Token**
- 统一配置 `baseURL`、`Content-Type`
- 支持 `get/post/put/delete`
- 支持**取消请求**，防止重复请求/页面切换卡顿
- 用泛型 `<T>` 给接口返回值做 TS 类型提示
