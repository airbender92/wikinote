-- 创建数据库
CREATE DATABASE testdb;

-- 使用数据库
USE testdb;

-- 查看当前选中的是哪个库
SELECT DATABASE();

--  查看所有库
SHOW DATABASES;

-- 创建学生表
CREATE TABLE student (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(20),
    age INT,
    gender VARCHAR(10)
);

-- 修改 testdb 库 student 表的 name 字段，限定为 20 位 utf8mb4 字符串、配套标准排序规则并设置为必填项
ALTER TABLE testdb.student MODIFY COLUMN name varchar(20) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci NOT NULL;
